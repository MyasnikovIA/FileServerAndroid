Создание keystore.p12 (openssl)

# 1. Приватный ключ + самоподписанный сертификат
openssl req -x509 -newkey rsa:2048 -nodes -days 3650 \
    -keyout key.pem -out cert.pem \
    -subj "/CN=fileserver.local/O=FileServer/C=RU"

# 2. Упаковка в PKCS12
openssl pkcs12 -export \
    -in cert.pem -inkey key.pem \
    -out keystore.p12 -name server \
    -passout pass:changeit

# 3. Копирование в каталог приложения (например, через adb)
adb push keystore.p12 /sdcard/Android/data/ru.miacomsoft.fileserver/files/certs/keystore.p12
